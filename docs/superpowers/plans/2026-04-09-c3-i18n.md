# C-3: i18n / WPML・Polylang 対応 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Tech Stack:** TypeScript 6.0.2, vitest 4.1.3, zod 4.3.6, pnpm monorepo

`packages/analyzer/src/i18n-detector.ts`:

```typescript
import type { WpPost } from "@wp-transfer/core";

export interface I18nDetectionResult {
  plugin: "wpml" | "polylang" | null;
  locales: string[];
  defaultLocale: string;
  postLocaleMap: Map<number, string>;
}

const LOCALE_PATTERN = /^[a-z]{2}(-[a-z]{2})?$/i;

function isValidLocale(value: string): boolean {
  return LOCALE_PATTERN.test(value);
}

/**
 * Detect i18n plugin usage (WPML or Polylang) from WXR parsed posts.
 *
 * WPML: stores language in meta["wpml_language"].
 * Polylang: stores language as a taxonomy term with domain="language".
 *
 * Returns detected plugin, locale list, default locale (most posts),
 * and a post-to-locale mapping.
 */
export function detectI18n(posts: WpPost[]): I18nDetectionResult {
  const postLocaleMap = new Map<number, string>();

  // Try WPML first (meta-based, faster)
  let isWpml = false;
  for (const post of posts) {
    const lang = post.meta["wpml_language"];
    if (typeof lang === "string" && lang) {
      if (isValidLocale(lang)) {
        postLocaleMap.set(post.id, lang);
        isWpml = true;
      }
    }
  }

  if (isWpml) {
    return buildResult("wpml", postLocaleMap);
  }

  // Try Polylang (terms-based)
  let isPolylang = false;
  for (const post of posts) {
    if (!post.terms) continue;
    const langTerm = post.terms.find((t) => t.domain === "language");
    if (langTerm && isValidLocale(langTerm.slug)) {
      postLocaleMap.set(post.id, langTerm.slug);
      isPolylang = true;
    }
  }

  if (isPolylang) {
    return buildResult("polylang", postLocaleMap);
  }

  return { plugin: null, locales: [], defaultLocale: "", postLocaleMap: new Map() };
}

function buildResult(
  plugin: "wpml" | "polylang",
  postLocaleMap: Map<number, string>,
): I18nDetectionResult {
  // Count posts per locale
  const localeCounts = new Map<string, number>();
  for (const locale of postLocaleMap.values()) {
    localeCounts.set(locale, (localeCounts.get(locale) ?? 0) + 1);
  }

  // Sort locales alphabetically
  const locales = [...localeCounts.keys()].sort();

  // Default = most posts
  let defaultLocale = "";
  let maxCount = 0;
  for (const [locale, count] of localeCounts) {
    if (count > maxCount) {
      maxCount = count;
      defaultLocale = locale;
    }
  }

  return { plugin, locales, defaultLocale, postLocaleMap };
}
```

```typescript
import type { ScaffoldFile } from "./blog-scaffold-generator.js";

export interface I18nScaffoldInput {
  locales: string[];
  defaultLocale: string;
}

export function generateI18nScaffold(input: I18nScaffoldInput): ScaffoldFile[] {
  const files: ScaffoldFile[] = [];

  files.push({ path: "middleware.ts", content: generateMiddleware(input) });
  files.push({ path: "i18n/config.ts", content: generateConfig(input) });
  files.push({ path: "app/[locale]/layout.tsx", content: generateLocaleLayout(input) });
  files.push({ path: "app/[locale]/page.tsx", content: generateLocalePage(input) });

  return files;
}

function generateMiddleware(input: I18nScaffoldInput): string {
  const localesList = input.locales.map((l) => `"${l}"`).join(", ");

  return `import { NextResponse, type NextRequest } from "next/server";

const locales = [${localesList}];
const defaultLocale = "${input.defaultLocale}";

function getLocaleFromHeaders(request: NextRequest): string {
  const acceptLanguage = request.headers.get("accept-language") ?? "";
  for (const locale of locales) {
    if (acceptLanguage.includes(locale)) {
      return locale;
    }
  }
  return defaultLocale;
}

function getLocaleFromPath(pathname: string): string | null {
  for (const locale of locales) {
    if (pathname.startsWith(\`/\${locale}/\`) || pathname === \`/\${locale}\`) {
      return locale;
    }
  }
  return null;
}

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Skip static files and API routes
  if (
    pathname.startsWith("/_next") ||
    pathname.startsWith("/api") ||
    pathname.includes(".")
  ) {
    return NextResponse.next();
  }

  const pathnameLocale = getLocaleFromPath(pathname);

  if (pathnameLocale) {
    // Valid locale in path — proceed
    const response = NextResponse.next();
    response.cookies.set("NEXT_LOCALE", pathnameLocale);
    return response;
  }

  // No locale in path — redirect
  const cookieLocale = request.cookies.get("NEXT_LOCALE")?.value;
  const locale =
    cookieLocale && locales.includes(cookieLocale)
      ? cookieLocale
      : getLocaleFromHeaders(request);

  return NextResponse.redirect(
    new URL(\`/\${locale}\${pathname}\`, request.url),
  );
}

export const config = {
  matcher: ["/((?!_next|api|.*\\\\..*).*)"],
};
`;
}

function generateConfig(input: I18nScaffoldInput): string {
  const localesList = input.locales.map((l) => `"${l}"`).join(", ");

  return `export const locales = [${localesList}] as const;
export const defaultLocale = "${input.defaultLocale}" satisfies Locale;

export type Locale = (typeof locales)[number];
`;
}

function generateLocaleLayout(input: I18nScaffoldInput): string {
  const localesList = input.locales.map((l) => `"${l}"`).join(", ");

  return `import type { ReactNode } from "react";
import { notFound } from "next/navigation";

const locales = [${localesList}];

export default async function LocaleLayout({
  children,
  params,
}: {
  children: ReactNode;
  params: Promise<{ locale: string }>;
}) {
  const { locale } = await params;

  if (!locales.includes(locale)) {
    notFound();
  }

  return (
    <html lang={locale}>
      <body>{children}</body>
    </html>
  );
}
`;
}

function generateLocalePage(input: I18nScaffoldInput): string {
  const localesList = input.locales.map((l) => `"${l}"`).join(", ");

  return `import Link from "next/link";

const locales = [${localesList}];

export default async function HomePage({
  params,
}: {
  params: Promise<{ locale: string }>;
}) {
  const { locale } = await params;

  return (
    <div>
      <h1>Welcome — {locale}</h1>
      <nav>
        <ul>
          {locales.map((l) => (
            <li key={l}>
              <Link href={\`/\${l}\`}>{l}</Link>
            </li>
          ))}
        </ul>
      </nav>
    </div>
  );
}
`;
}
```

```typescript
import { describe, it, expect } from "vitest";
import { createReadStream } from "node:fs";
import { resolve } from "node:path";
import { parseWxr } from "@wp-transfer/wxr-parser";
import { detectI18n, generateI18nScaffold } from "../src/index.js";

describe("E2E: i18n WXR → Detection → Scaffold", () => {

    expect(config.content).toContain('"en"');
    expect(config.content).toContain('"ja"');

    for (const file of files) {
      expect(file.content.length).toBeGreaterThan(0);
    }

      locales: i18n.locales,

      expect(file.content.length).toBeGreaterThan(0);
```
